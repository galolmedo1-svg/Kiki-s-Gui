local function checkBackdoor()
    local suspiciousExactNames = {"serverside", "backdoor", "topk3k", "runscript", "require"} 
    for _, v in pairs(game:GetDescendants()) do
        if v:IsA("RemoteEvent") or v:IsA("RemoteFunction") then
            local name = string.lower(v.Name)
            for _, sName in pairs(suspiciousExactNames) do
                if name == sName then foundBackdoorRemote = v; return true end
            end
            if v.Parent and v.Parent.Name == "JointsService" then foundBackdoorRemote = v; return true end
        end
    end
    return false
end

local function findPartByNameOrPartial(partialName)
    local target = nil
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") or v:IsA("Model") then
            if string.find(string.lower(v.Name), string.lower(partialName)) then
                target = v:IsA("Model") and (v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")) or v
                if target then break end
            end
        end
    end
    return target
end

            local function getCFrameFromTarget(targetName)
                if targetName == "Spawn" then return workspace:FindFirstChildWhichIsA("SpawnLocation", true) and workspace:FindFirstChildWhichIsA("SpawnLocation", true).CFrame or CFrame.new(0,5,0) end
                local p = Players:FindFirstChild(targetName)
                if p and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then return p.Character.HumanoidRootPart.CFrame end
                local obj = findPartByNameOrPartial(targetName)
                if obj then return obj.CFrame end
                return nil
            end

            local function sendItemTo(itemName, targetName)
                local item = findPartByNameOrPartial(itemName)
                local targetCFrame = getCFrameFromTarget(targetName)
                if item and targetCFrame then
                    if item:IsA("Model") then
                        item:SetPrimaryPartCFrame(targetCFrame)
                    else
                        item.CFrame = targetCFrame
                    end
                    Rayfield:Notify({Title="Success", Content="Item " .. itemName .. " sent to " .. targetName, Duration=3})
                else
                    Rayfield:Notify({Title="Error", Content="Item or destination not found.", Duration=3})
                end
            end

                end
            end

    TabCustom:CreateSection("Language")
    local LangDropdown
    LangDropdown = TabCustom:CreateDropdown({
        Name = "Language / Idioma", Options = langOptionsMap[currentLanguage] or langOptionsMap["English"], CurrentOption = langOptionsMap[currentLanguage][1], MultipleOptions = false,
        Callback = function(Option)
            local sel = Option[1] or ""
            if string.find(sel, "Español") then currentLanguage = "Español"
            elseif string.find(sel, "English") then currentLanguage = "English"
            elseif string.find(sel, "Português") then currentLanguage = "Português"
            elseif string.find(sel, "Guarani") then currentLanguage = "Guarani"
            elseif string.find(sel, "한국어") then currentLanguage = "한국어" end
            pcall(function() LangDropdown:Refresh(langOptionsMap[currentLanguage], true) end)
        end
    })

    TabCustom:CreateButton({ Name = "Reset message text transparency", Callback = function() currentMsgTextTrans = defaultMsgTextTrans end })

    TabCustom:CreateButton({ Name = "Reset all panel colors", Callback = function() showConfirmationModal("all panel colors", function() currentPanelBgColor = defaultPanelBgColor currentPanelTextColor = defaultPanelTextColor pcall(function() Rayfield:ModifyTheme({ BackgroundColor = defaultPanelBgColor, TextColor = defaultPanelTextColor }) end) end) end })

    TabCustom:CreateButton({ Name = "Reset everything on panel (includes buttons)", Callback = function() showConfirmationModal("everything on panel (includes buttons)", function() currentPanelBgColor = defaultPanelBgColor currentPanelTextColor = defaultPanelTextColor currentBtnBgColor = defaultBtnBgColor currentBtnTextColor = defaultBtnTextColor pcall(function() Rayfield:ModifyTheme({ BackgroundColor = defaultPanelBgColor, TextColor = defaultPanelTextColor, ElementBackground = defaultBtnBgColor, SecondaryText = defaultBtnTextColor }) end) end) end })
    TabCustom:CreateButton({ Name = "Reset all message transparencies", Callback = function() showConfirmationModal("all message transparencies", function() currentMsgBgTrans = defaultMsgBgTrans currentMsgTextTrans = defaultMsgTextTrans end) end })
    TabCustom:CreateButton({ Name = "Reset everything", Callback = function() showConfirmationModal("everything", function() currentPanelBgColor = defaultPanelBgColor currentPanelTextColor = defaultPanelTextColor currentBtnBgColor = defaultBtnBgColor currentBtnTextColor = defaultBtnTextColor currentMsgBgColor = defaultMsgBgColor currentMsgTextColor = defaultMsgTextColor currentMsgBgTrans = defaultMsgBgTrans currentMsgTextTrans = defaultMsgTextTrans currentMsgDuration = defaultMsgDuration currentLanguage = "English" pcall(function() Rayfield:ModifyTheme({ BackgroundColor = defaultPanelBgColor, TextColor = defaultPanelTextColor, ElementBackground = defaultBtnBgColor, SecondaryText = defaultBtnTextColor }) if LangDropdown then LangDropdown:Refresh(langOptionsMap["English"], true) end end) end) end })
